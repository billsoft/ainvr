cmake_minimum_required(VERSION 3.22)
if(CMAKE_BINARY_DIR STREQUAL "${CMAKE_SOURCE_DIR}/build")
  message(FATAL_ERROR "license-service/build is reserved for deployment. Configure in ../out/build/<purpose> instead.")
endif()
project(ainvr_license_service LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json 3 REQUIRED)
find_package(Threads REQUIRED)
find_package(httplib QUIET)
if(NOT TARGET httplib::httplib)
  find_library(HTTPLIB_LIB NAMES cpp-httplib httplib REQUIRED)
  find_path(HTTPLIB_INCLUDE httplib.h REQUIRED)
  add_library(httplib::httplib SHARED IMPORTED)
  set_target_properties(httplib::httplib PROPERTIES IMPORTED_LOCATION "${HTTPLIB_LIB}" INTERFACE_INCLUDE_DIRECTORIES "${HTTPLIB_INCLUDE}" INTERFACE_LINK_LIBRARIES "OpenSSL::SSL;OpenSSL::Crypto;Threads::Threads")
endif()
add_library(license_core STATIC
  src/encoding.cpp src/crypto.cpp src/fingerprint.cpp
  src/state_store.cpp src/installation.cpp src/signed_license.cpp src/license_manager.cpp src/config.cpp
  src/license_http_app.cpp src/uds_server.cpp src/clock.cpp)
target_include_directories(license_core PUBLIC src)
set(AINVR_ISSUER_KEY_ID "" CACHE STRING "AINVR issuer key identifier")
set(AINVR_ISSUER_PUBLIC_KEY_HEX "" CACHE STRING "AINVR Ed25519 public key (64 hex characters)")
if(AINVR_ISSUER_KEY_ID OR AINVR_ISSUER_PUBLIC_KEY_HEX)
  string(LENGTH "${AINVR_ISSUER_PUBLIC_KEY_HEX}" PUBLIC_KEY_LENGTH)
  if(NOT PUBLIC_KEY_LENGTH EQUAL 64 OR NOT AINVR_ISSUER_PUBLIC_KEY_HEX MATCHES "^[0-9a-fA-F]+$" OR NOT AINVR_ISSUER_KEY_ID MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "Supply a valid AINVR issuer identifier and Ed25519 public key")
  endif()
  target_compile_definitions(license_core PRIVATE
    AINVR_ISSUER_KEY_ID="${AINVR_ISSUER_KEY_ID}"
    AINVR_ISSUER_PUBLIC_KEY_HEX="${AINVR_ISSUER_PUBLIC_KEY_HEX}")
endif()
target_link_libraries(license_core PUBLIC OpenSSL::Crypto yaml-cpp nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(license_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(license_service src/main.cpp)
target_link_libraries(license_service PRIVATE license_core httplib::httplib)
option(BUILD_TESTING "Build isolated tests with synthetic credentials" ON)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  enable_testing()
  add_executable(license_tests tests/crypto_test.cpp tests/manager_test.cpp tests/diagnostics_test.cpp tests/v3_test.cpp)
  target_link_libraries(license_tests PRIVATE license_core GTest::gtest_main)
  add_test(NAME license_tests COMMAND license_tests)
endif()

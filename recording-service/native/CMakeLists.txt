cmake_minimum_required(VERSION 3.22)
project(ainvr_recording_frame LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
include(../../ai-service/cmake/ffmpeg_runtime.cmake)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
set(PLATFORM ../../ai-service/src/platform)
add_executable(ainvr_recording_frame recording_frame.cpp ${PLATFORM}/vpu_decoder.cpp ${PLATFORM}/media_memory.cpp ${PLATFORM}/jpeg_encoder.cpp ${PLATFORM}/gles_preprocess.cpp)
target_include_directories(ainvr_recording_frame PRIVATE ${PLATFORM} /usr/hobot/include)
foreach(name multimedia hbmem EGL GLESv2 gbm)
 find_library(HB_${name} ${name} PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
 target_link_libraries(ainvr_recording_frame PRIVATE ${HB_${name}})
endforeach()
target_link_libraries(ainvr_recording_frame PRIVATE PkgConfig::WAYLAND PkgConfig::AVFORMAT)
set_target_properties(ainvr_recording_frame PROPERTIES BUILD_RPATH /usr/hobot/lib)

cmake_minimum_required(VERSION 3.22)
project(ainvr_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
set(TOKENIZERS_SOURCE /mnt/ainvr-ssd/llm-build/third_party/tokenizers-cpp CACHE PATH "Pinned tokenizers-cpp c586c52 source")
# Qwen uses HF JSON only. Build that pinned adapter without downloading unused
# SentencePiece/Abseil dependencies or modifying third-party sources.
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(TOKENIZERS_RUST_LIB ${CMAKE_BINARY_DIR}/rust/release/libtokenizers_c.a)
add_custom_command(OUTPUT ${TOKENIZERS_RUST_LIB}
  COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${CMAKE_BINARY_DIR}/rust
          ${CARGO_EXECUTABLE} build --release --locked --manifest-path ${TOKENIZERS_SOURCE}/rust/Cargo.toml
  DEPENDS ${TOKENIZERS_SOURCE}/rust/Cargo.lock ${TOKENIZERS_SOURCE}/rust/Cargo.toml)
add_custom_target(tokenizers_rust DEPENDS ${TOKENIZERS_RUST_LIB})
add_library(tokenizers_cpp STATIC ${TOKENIZERS_SOURCE}/src/huggingface_tokenizer.cc)
add_dependencies(tokenizers_cpp tokenizers_rust)
target_include_directories(tokenizers_cpp PUBLIC ${TOKENIZERS_SOURCE}/include)
target_link_libraries(tokenizers_cpp PUBLIC ${TOKENIZERS_RUST_LIB} Threads::Threads ${CMAKE_DL_LIBS})
find_library(HB_DNN dnn PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
find_library(HB_UCP hbucp PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
add_library(ainvr_llm_engine src/bpu_program.cpp src/engine.cpp)
target_include_directories(ainvr_llm_engine PUBLIC src ${TOKENIZERS_SOURCE}/include /usr/include/hobot)
target_link_libraries(ainvr_llm_engine PUBLIC tokenizers_cpp ${HB_DNN} ${HB_UCP} OpenSSL::Crypto nlohmann_json::nlohmann_json Threads::Threads)
add_executable(ainvr_llm_probe src/probe.cpp)
target_link_libraries(ainvr_llm_probe PRIVATE ainvr_llm_engine)
set_target_properties(ainvr_llm_probe PROPERTIES BUILD_RPATH /usr/hobot/lib)
add_executable(ainvr_llm_layout_probe tests/layout_probe.cpp)
target_include_directories(ainvr_llm_layout_probe PRIVATE src)
target_link_libraries(ainvr_llm_layout_probe PRIVATE nlohmann_json::nlohmann_json)
find_package(PkgConfig REQUIRED)
include(../ai-service/cmake/ffmpeg_runtime.cmake)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
find_library(HB_MEDIA multimedia PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
find_library(HB_MEM hbmem PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
find_library(HB_EGL EGL PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
find_library(HB_GLES GLESv2 PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
find_library(HB_GBM gbm PATHS /usr/hobot/lib REQUIRED NO_DEFAULT_PATH)
add_library(ainvr_llm_image
  ../ai-service/src/platform/vpu_decoder.cpp
  ../ai-service/src/platform/media_memory.cpp
  ../ai-service/src/platform/jpeg_encoder.cpp
  ../ai-service/src/platform/gles_preprocess.cpp)
target_include_directories(ainvr_llm_image PUBLIC ../ai-service/src/platform PRIVATE /usr/hobot/include)
target_link_libraries(ainvr_llm_image PUBLIC ${HB_MEDIA} ${HB_MEM} ${HB_EGL} ${HB_GLES} ${HB_GBM} PkgConfig::WAYLAND PkgConfig::AVFORMAT)
add_executable(ainvr_llm_image_probe src/image_probe.cpp)
target_link_libraries(ainvr_llm_image_probe PRIVATE ainvr_llm_image)
set_target_properties(ainvr_llm_image_probe PROPERTIES BUILD_RPATH /usr/hobot/lib)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
add_executable(ainvr_llm_service src/service.cpp)
target_link_libraries(ainvr_llm_service PRIVATE ainvr_llm_engine ainvr_llm_image PkgConfig::HTTP)
set_target_properties(ainvr_llm_service PROPERTIES BUILD_RPATH /usr/hobot/lib)

include(CTest)
add_executable(ainvr_llm_configuration_state_test tests/configuration_state_test.cpp)
target_include_directories(ainvr_llm_configuration_state_test PRIVATE src)
target_link_libraries(ainvr_llm_configuration_state_test PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(ainvr_llm_configuration_state_test PRIVATE -UNDEBUG)
add_test(NAME llm_configuration_state COMMAND ainvr_llm_configuration_state_test)

add_executable(ainvr_llm_request_test tests/request_test.cpp)
target_include_directories(ainvr_llm_request_test PRIVATE src)
target_link_libraries(ainvr_llm_request_test PRIVATE nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_options(ainvr_llm_request_test PRIVATE -UNDEBUG)
add_test(NAME llm_request_validation COMMAND ainvr_llm_request_test)

target_compile_options(ainvr_llm_service PRIVATE -idirafter /usr/hobot/include)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(ainvr_llm_reservation_probe tests/reservation_probe.cpp)
target_include_directories(ainvr_llm_reservation_probe PRIVATE src)
target_link_libraries(ainvr_llm_reservation_probe PRIVATE nlohmann_json::nlohmann_json)
add_test(NAME llm_residency_lock COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_residency_lock.py $<TARGET_FILE:ainvr_llm_reservation_probe>)

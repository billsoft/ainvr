cmake_minimum_required(VERSION 3.16)
project(ainvr_analytics_tests LANGUAGES CXX)
enable_testing()
foreach(suite core tracker config)
  add_executable(analytics_${suite}_test ../ai_port/analytics_${suite}_test.cpp)
  target_compile_features(analytics_${suite}_test PRIVATE cxx_std_17)
  target_compile_options(analytics_${suite}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME analytics_${suite} COMMAND analytics_${suite}_test)
  set_tests_properties(analytics_${suite} PROPERTIES TIMEOUT 30)
endforeach()

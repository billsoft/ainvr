# The board's system cpp-httplib 0.10.3 corrupts valid multipart requests when
# TCP splits a boundary prefix. Pin the upstream header for this service only;
# never mix this header with the old system .so or alter existing services.
set(AINVR_HTTP_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/cpp-httplib")
file(SHA256 "${AINVR_HTTP_INCLUDE}/httplib.h" AINVR_HTTP_SHA)
if(NOT AINVR_HTTP_SHA STREQUAL "9021001a719768893a8a6e4051c56107690e8b8e49b41e5f903534fa19ae9afb")
  message(FATAL_ERROR "Pinned HTTP dependency checksum mismatch")
endif()
add_library(ainvr_search_http_transport INTERFACE)
target_include_directories(ainvr_search_http_transport SYSTEM INTERFACE "${AINVR_HTTP_INCLUDE}")
target_link_libraries(ainvr_search_http_transport INTERFACE Threads::Threads)
